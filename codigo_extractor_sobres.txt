    <script>
        // --- CONFIGURACIÓN E INICIALIZACIÓN ---
        const fileInput = document.getElementById('fileInput');
        const btnProcesar = document.getElementById('btnProcesar');
        const btnExportar = document.getElementById('btnExportar');
        const apiKeyInput = document.getElementById('apiKey');
        const panelResultados = document.getElementById('panelResultados');
        const tbodyDatos = document.getElementById('tbodyDatos');
        const logConsole = document.getElementById('logConsole');
        const contadorProgreso = document.getElementById('contadorProgreso');

        let datosDataset = []; // Almacén de objetos extraídos

        // Habilitar botón si hay archivos
        fileInput.addEventListener('change', () => {
            btnProcesar.disabled = fileInput.files.length === 0;
        });

        // Logger profesional
        function log(mensaje, tipo = 'info') {
            logConsole.classList.remove('hidden');
            const color = tipo === 'error' ? 'text-red-400' : (tipo === 'success' ? 'text-cyan-400' : 'text-emerald-400');
            logConsole.innerHTML += `<div class="${color}">[${new Date().toLocaleTimeString()}] ${mensaje}</div>`;
            logConsole.scrollTop = logConsole.scrollHeight;
        }

        // --- PIPELINE DE INTELIGENCIA ARTIFICIAL (GEMINI) ---
        btnProcesar.addEventListener('click', async () => {
            const apiKey = apiKeyInput.value.trim();
            if (!apiKey) {
                alert('Por favor, introduce una API Key válida de Gemini.');
                return;
            }

            const files = fileInput.files;
            datosDataset = []; // Reset
            tbodyDatos.innerHTML = '';
            panelResultados.classList.remove('hidden');
            
            log(`Iniciando procesamiento por lotes para ${files.length} archivo(s)...`);
            btnProcesar.disabled = true;

            for (let i = 0; i < files.length; i++) {
                contadorProgreso.innerText = `${i + 1}/${files.length} Procesando...`;
                try {
                    log(`Procesando archivo: ${files[i].name}...`);
                    const base64Data = await convertToBase64(files[i]);
                    const cleanBase64 = base64Data.split(',')[1];
                    const mimeType = files[i].type;

                    const resultadoIa = await consultarGemini(cleanBase64, mimeType, apiKey);
                    const datosEstructurados = JSON.parse(resultadoIa);
                    
                    datosDataset.push(datosEstructurados);
                    agregarFilaTabla(datosEstructurados);
                    log(`Archivo ${files[i].name} procesado con éxito.`, 'success');
                } catch (error) {
                    log(`Error procesando ${files[i].name}: \${error.message}`, 'error');
                }
            }

            contadorProgreso.innerText = `\${files.length}/\${files.length} Finalizado`;
            contadorProgreso.className = "text-xs bg-emerald-100 text-emerald-800 px-2.5 py-1 rounded-full font-medium";
            btnProcesar.disabled = false;
            log('Procesamiento de lote completo.', 'success');
        });

        // Conversor Asíncrono de Archivos a Base64
        function convertToBase64(file) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.readAsDataURL(file);
                reader.onload = () => resolve(reader.result);
                reader.onerror = error => reject(error);
            });
        }

        // --- LLAMADA ESTRUCTURADA A GEMINI 2.5 FLASH ---
        async function consultarGemini(base64Image, mimeType, apiKey) {
            const prompt = `Analiza detalladamente esta foto de un sobre de revelado fotográfico Fujifilm (Mi Foto Jumbo). Extrae rigurosamente la información manuscrita e impresa.
            Devuelve EXCLUSIVAMENTE un objeto JSON válido (sin markdown, sin bloques de código \`\`\`, solo texto plano con la estructura JSON) con los siguientes campos:
            {
                "no_sobre": "Texto o número en 'nº de sobre'",
                "no_orden": "Texto o número manuscrito en 'nº de orden / ID' (ej: '#642194')",
                "cliente": "Nombre completo manuscrito en 'Nombre cliente'",
                "telefono": "Número en 'Teléfono'",
                "codigo_mini": "Número pequeño impreso en la etiqueta pegada (ej: '6246')",
                "tienda": "Nombre impreso o cuño de la sucursal en 'Nombre tienda' (ej: 'Mi Foto Jumbo Alto Las Condes')",
                "formatos": "Formatos marcados con X o números (ej: 'Otro: X')",
                "servicios": "Servicios marcados como Impresión con cantidad, Digital, Traspaso DVD o Pendrive (ej: 'Impresión: 36')",
                "observaciones": "Texto manuscrito en la sección de 'Observaciones' al final"
            }
            Si algún dato no es visible o está vacío, pon una cadena de texto vacía. Sé exacto con la transcripción.`;

            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=\${apiKey}`;

            const response = await fetch(url, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{
                        parts: [
                            { text: prompt },
                            { inlineData: { mimeType: mimeType, data: base64Image } }
                        ]
                    }],
                    generationConfig: {
                        responseMimeType: "application/json"
                    }
                })
            });

            if (!response.ok) {
                const errData = await response.json();
                throw new Error(errData.error?.message || 'Error desconocido en API');
            }

            const resData = await response.json();
            return resData.candidates[0].content.parts[0].text;
        }

        // --- INTERFAZ DE USUARIO Y EXPORTACIÓN ---
        function agregarFilaTabla(data) {
            const tr = document.createElement('tr');
            tr.className = "hover:bg-slate-50 transition-colors";
            tr.innerHTML = `
                <td class="px-4 py-3 font-medium text-slate-900">\${data.no_sobre || ''}</td>
                <td class="px-4 py-3 font-mono text-blue-600">\${data.no_orden || ''}</td>
                <td class="px-4 py-3">\${data.cliente || ''}</td>
                <td class="px-4 py-3">\${data.telefono || ''}</td>
                <td class="px-4 py-3 text-xs bg-slate-100 text-center font-bold rounded-sm">\${data.codigo_mini || ''}</td>
                <td class="px-4 py-3">\${data.tienda || ''}</td>
                <td class="px-4 py-3">\${data.formatos || ''}</td>
                <td class="px-4 py-3">\${data.servicios || ''}</td>
                <td class="px-4 py-3 italic">\${data.observaciones || ''}</td>
            `;
            tbodyDatos.appendChild(tr);
        }

        // Conversión y descarga a Excel utilizando SheetJS
        btnExportar.addEventListener('click', () => {
            if (datosDataset.length === 0) return;

            const filasExcel = datosDataset.map(item => ({
                'N° Sobre': item.no_sobre,
                'N° Orden / ID': item.no_orden,
                'Cliente': item.cliente,
                'Teléfono': item.telefono,
                'Código Mini': item.codigo_mini,
                'Tienda': item.tienda,
                'Formatos Seleccionados': item.formatos,
                'Servicios / Cantidad': item.servicios,
                'Observaciones': item.observaciones
            }));

            const hoja = XLSX.utils.json_to_sheet(filasExcel);
            const libro = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(libro, hoja, "Datos Sobres");

            XLSX.writeFile(libro, \`Sobres_Procesados_\${new Date().toISOString().slice(0,10)}.xlsx\`);
            log('Excel generado y descargado con éxito.', 'success');
        });
    </script>
